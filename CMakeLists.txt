cmake_minimum_required(VERSION 3.22)
project(M45DriveWatcher VERSION 0.1.4 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets DBus)
add_library(m45-drive-watcher-core src/monitor.cpp)
target_include_directories(m45-drive-watcher-core PUBLIC src)
target_link_libraries(m45-drive-watcher-core PUBLIC Qt6::Core)
target_compile_options(m45-drive-watcher-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(m45-drive-watcher src/main.cpp src/window.cpp)
target_compile_definitions(m45-drive-watcher PRIVATE M45_DRIVE_WATCHER_VERSION="${PROJECT_VERSION}")
target_link_libraries(m45-drive-watcher PRIVATE m45-drive-watcher-core Qt6::Widgets Qt6::DBus)
target_compile_options(m45-drive-watcher PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS m45-drive-watcher RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS collector/collect.py DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/m45-drive-watcher RENAME collect)
install(FILES packaging/m45-drive-watcher-collector.service packaging/m45-drive-watcher-collector.timer DESTINATION lib/systemd/system)
install(FILES packaging/com.m45software.DriveWatcher.desktop DESTINATION share/applications)
install(FILES packaging/com.m45software.DriveWatcher.desktop DESTINATION /etc/xdg/autostart)
install(FILES packaging/m45-drive-watcher.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES README.md LICENSE DESTINATION share/doc/m45-drive-watcher)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(test-monitor tests/test_monitor.cpp)
  target_link_libraries(test-monitor PRIVATE m45-drive-watcher-core Qt6::Test)
  add_test(NAME monitor COMMAND test-monitor)
  add_executable(test-window tests/test_window.cpp src/window.cpp)
  target_link_libraries(test-window PRIVATE m45-drive-watcher-core Qt6::Test Qt6::Widgets Qt6::DBus)
  add_test(NAME window COMMAND test-window)
  set_tests_properties(window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME collector COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
  set_tests_properties(collector PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME m45-drive-watcher)
set(CPACK_PACKAGE_CONTACT "M45 Software")
set(CPACK_DEBIAN_PACKAGE_CONFLICTS "drivewatch")
set(CPACK_DEBIAN_PACKAGE_REPLACES "drivewatch")
set(CPACK_DEBIAN_PACKAGE_PROVIDES "drivewatch")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "KDE tray monitor for changing drive health and temperatures")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_PACKAGE_SECTION utils)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3 (>= 3.10), smartmontools (>= 7.3), systemd, init-system-helpers, qt6-qpa-plugins")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "qt6-wayland")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_SOURCE_DIR}/packaging/postinst;${CMAKE_SOURCE_DIR}/packaging/prerm;${CMAKE_SOURCE_DIR}/packaging/postrm")
set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION TRUE)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
include(CPack)
