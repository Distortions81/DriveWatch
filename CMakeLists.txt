cmake_minimum_required(VERSION 3.22)
project(DriveWatch VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
set(CMAKE_INSTALL_DEFAULT_DIRECTORY_PERMISSIONS OWNER_READ OWNER_WRITE OWNER_EXECUTE GROUP_READ GROUP_EXECUTE WORLD_READ WORLD_EXECUTE)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets DBus)
add_library(drivewatch-core src/monitor.cpp)
target_include_directories(drivewatch-core PUBLIC src)
target_link_libraries(drivewatch-core PUBLIC Qt6::Core)
target_compile_options(drivewatch-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(drivewatch src/main.cpp src/window.cpp)
target_link_libraries(drivewatch PRIVATE drivewatch-core Qt6::Widgets Qt6::DBus)
target_compile_options(drivewatch PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS drivewatch RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(PROGRAMS collector/collect.py DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/drivewatch RENAME collect)
install(FILES packaging/drivewatch-collector.service packaging/drivewatch-collector.timer DESTINATION lib/systemd/system)
install(FILES packaging/io.github.drivewatch.desktop DESTINATION share/applications)
install(FILES packaging/io.github.drivewatch.desktop DESTINATION /etc/xdg/autostart)
install(FILES packaging/drivewatch.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES README.md LICENSE DESTINATION share/doc/drivewatch)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(test-monitor tests/test_monitor.cpp)
  target_link_libraries(test-monitor PRIVATE drivewatch-core Qt6::Test)
  add_test(NAME monitor COMMAND test-monitor)
  add_executable(test-window tests/test_window.cpp src/window.cpp)
  target_link_libraries(test-window PRIVATE drivewatch-core Qt6::Test Qt6::Widgets Qt6::DBus)
  add_test(NAME window COMMAND test-window)
  set_tests_properties(window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  add_test(NAME collector COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
  set_tests_properties(collector PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
endif()
set(CPACK_GENERATOR DEB)
set(CPACK_PACKAGE_NAME drivewatch)
set(CPACK_PACKAGE_CONTACT "DriveWatch contributors")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "KDE tray monitor for changing drive health and temperatures")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_DEBIAN_PACKAGE_SECTION utils)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "python3 (>= 3.10), smartmontools (>= 7.3), systemd, init-system-helpers, qt6-qpa-plugins")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "qt6-wayland")
set(CPACK_DEBIAN_PACKAGE_CONTROL_EXTRA "${CMAKE_SOURCE_DIR}/packaging/postinst;${CMAKE_SOURCE_DIR}/packaging/prerm;${CMAKE_SOURCE_DIR}/packaging/postrm")
set(CPACK_DEBIAN_PACKAGE_CONTROL_STRICT_PERMISSION TRUE)
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
include(CPack)
